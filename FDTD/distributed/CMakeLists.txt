cmake_minimum_required(VERSION 3.1...3.10)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(openEMS_domain_runtime CXX)
  enable_testing()
endif()

add_library(openEMS_domain_runtime STATIC domain_plan.cpp runtime.cpp cpu_executor.cpp)
set_target_properties(openEMS_domain_runtime PROPERTIES CXX_STANDARD 11 POSITION_INDEPENDENT_CODE ON)
target_include_directories(openEMS_domain_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../..)

option(OPENEMS_DOMAIN_TESTS "Build experimental domain-runtime tests" OFF)
if(OPENEMS_DOMAIN_TESTS)
  enable_testing()
  add_executable(test_domain_runtime ../../TESTSUITE/distributed/test_runtime.cpp)
  target_link_libraries(test_domain_runtime openEMS_domain_runtime)
  set_target_properties(test_domain_runtime PROPERTIES CXX_STANDARD 11)
  add_test(NAME domain_runtime COMMAND test_domain_runtime)
  set_tests_properties(domain_runtime PROPERTIES TIMEOUT 40)
  add_executable(test_domain_fields ../../TESTSUITE/distributed/test_fields.cpp)
  target_link_libraries(test_domain_fields openEMS_domain_runtime)
  set_target_properties(test_domain_fields PROPERTIES CXX_STANDARD 11)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(openEMS_domain_runtime PRIVATE -ffp-contract=off)
    target_compile_options(test_domain_fields PRIVATE -ffp-contract=off)
  endif()
  add_test(NAME domain_fields COMMAND test_domain_fields)
  set_tests_properties(domain_fields PROPERTIES TIMEOUT 40)
endif()

option(OPENEMS_DOMAIN_MPI "Build the optional experimental MPI transport" OFF)
if(OPENEMS_DOMAIN_MPI)
  if(CMAKE_VERSION VERSION_LESS 3.9)
    message(FATAL_ERROR "The optional MPI adapter requires CMake 3.9 or newer")
  endif()
  find_package(MPI REQUIRED COMPONENTS CXX)
  add_library(openEMS_domain_mpi STATIC mpi_transport.cpp)
  set_target_properties(openEMS_domain_mpi PROPERTIES CXX_STANDARD 11 POSITION_INDEPENDENT_CODE ON)
  target_link_libraries(openEMS_domain_mpi PUBLIC openEMS_domain_runtime MPI::MPI_CXX)
  if(OPENEMS_DOMAIN_TESTS)
    add_executable(test_domain_mpi ../../TESTSUITE/distributed/test_mpi.cpp)
    target_link_libraries(test_domain_mpi openEMS_domain_mpi)
    set_target_properties(test_domain_mpi PROPERTIES CXX_STANDARD 11)
    foreach(n 1 2 3)
      add_test(NAME domain_mpi_${n} COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${n}
        ${MPIEXEC_PREFLAGS} $<TARGET_FILE:test_domain_mpi> ${MPIEXEC_POSTFLAGS})
      set_tests_properties(domain_mpi_${n} PROPERTIES TIMEOUT 60)
    endforeach()
    add_test(NAME domain_mpi_wrong_epoch COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} 2
      ${MPIEXEC_PREFLAGS} $<TARGET_FILE:test_domain_mpi> --fail-epoch ${MPIEXEC_POSTFLAGS})
    set_tests_properties(domain_mpi_wrong_epoch PROPERTIES TIMEOUT 15 WILL_FAIL TRUE)
  endif()
endif()

option(OPENEMS_DOMAIN_HIP "Build the experimental HIP domain executor" OFF)
if(OPENEMS_DOMAIN_HIP)
  if(NOT ENABLE_HIP)
    message(FATAL_ERROR "Build the HIP executor through the full openEMS project with ENABLE_HIP")
  endif()
  set(DOMAIN_HIP_SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/../hip/hip_domain_executor.hip)
  if(OPENEMS_HIP_AS_CUDA)
    set_source_files_properties(${DOMAIN_HIP_SOURCE} PROPERTIES LANGUAGE CUDA COMPILE_OPTIONS "-x;cu")
  endif()
  add_library(openEMS_domain_hip STATIC ${DOMAIN_HIP_SOURCE})
  target_link_libraries(openEMS_domain_hip PUBLIC openEMS_domain_runtime)
  if(OPENEMS_HIP_AS_CUDA)
    target_include_directories(openEMS_domain_hip PRIVATE ${HIP_INCLUDE_DIR})
    target_compile_definitions(openEMS_domain_hip PRIVATE __HIP_PLATFORM_NVIDIA__=1)
  elseif(CMAKE_HIP_PLATFORM STREQUAL "amd")
    target_link_libraries(openEMS_domain_hip PRIVATE ${HIP_RUNTIME_LIBRARY})
  endif()
  if(OPENEMS_DOMAIN_TESTS)
    add_executable(test_domain_hip ../../TESTSUITE/distributed/test_hip.cpp)
    target_link_libraries(test_domain_hip openEMS_domain_hip)
    if(OPENEMS_HIP_AS_CUDA)
      target_include_directories(test_domain_hip PRIVATE ${HIP_INCLUDE_DIR})
      target_compile_definitions(test_domain_hip PRIVATE __HIP_PLATFORM_NVIDIA__=1)
      set_source_files_properties(../../TESTSUITE/distributed/test_hip.cpp PROPERTIES LANGUAGE CUDA)
    else()
      set_source_files_properties(../../TESTSUITE/distributed/test_hip.cpp PROPERTIES LANGUAGE HIP)
    endif()
  endif()
endif()

if(OPENEMS_DOMAIN_TESTS AND TARGET openEMS_domain_mpi AND TARGET openEMS_domain_hip)
  add_executable(test_domain_mpi_hip ../../TESTSUITE/distributed/test_mpi_hip.cpp)
  target_compile_definitions(test_domain_mpi_hip PRIVATE OPENEMS_DOMAIN_TEST_HIP=1)
  target_link_libraries(test_domain_mpi_hip openEMS_domain_mpi openEMS_domain_hip)
  add_executable(test_domain_mpi_cpu ../../TESTSUITE/distributed/test_mpi_hip.cpp)
  target_link_libraries(test_domain_mpi_cpu openEMS_domain_mpi)
endif()
